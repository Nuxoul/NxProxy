# Standalone harness: copy this file into a scratch directory as CMakeLists.txt.
# cmake -S <scratch-source> -B <scratch-build> -DNXPROXY_SOURCE=D:/Code/NxProxy
cmake_minimum_required(VERSION 3.20)
project(CoreRestartTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT NXPROXY_SOURCE)
    message(FATAL_ERROR "Supply -DNXPROXY_SOURCE=<repository root>")
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)
find_package(Threads REQUIRED)
add_executable(core_restart_policy "${NXPROXY_SOURCE}/tests/core_restart_policy.cpp")
target_include_directories(core_restart_policy PRIVATE "${NXPROXY_SOURCE}")
target_link_libraries(core_restart_policy PRIVATE Threads::Threads)
add_executable(core_restart_process
    "${NXPROXY_SOURCE}/tests/core_restart_process.cpp"
    "${NXPROXY_SOURCE}/src/sys/Process.cpp")
target_include_directories(core_restart_process BEFORE PRIVATE
    "${NXPROXY_SOURCE}/tests/core_restart_stubs" "${NXPROXY_SOURCE}")
target_link_libraries(core_restart_process PRIVATE Qt6::Core Qt6::Widgets Threads::Threads)
enable_testing()
add_test(NAME core_restart_policy COMMAND core_restart_policy)
add_test(NAME core_restart_process COMMAND core_restart_process)
set_tests_properties(core_restart_process PROPERTIES TIMEOUT 30)
